\section{Introduction}

\begin{frame}{High schools may shape consequential choices beyond achievement}
\begin{wideitemize}
  \item School quality is usually summarized by test scores, but educational production is not one-dimensional. \parencite{jackson_what_2018,beuermann_what_2023,rose_effects_2022}

  \item There are important gaps in application and enrollment behavior across groups even when conditioning on achievement. \parencite{hoxby_what_2015,dynarski_closing_2021,hastings_informed_2016,chetty_diversifying_2025}

  \item Returns differ substantially across institutions and especially across fields. \parencite{kirkeboen_field_2016}
\end{wideitemize}
\end{frame}

\begin{frame}{The paper asks whether schools causally shape college choices}
\begin{alertblock}{Main question}
Do high schools causally shape college choices?
\end{alertblock}

\begin{wideitemize}
  \item If so, are those effects fully predicted by achievement value added?
  \item Do the effects vary with students' baseline achievement?
\end{wideitemize}

\key{Answering these questions requires separating school effects from student sorting.}
\end{frame}

\begin{comment}
\begin{frame}{College choices need not reveal fixed preferences}
\begin{columns}[T,onlytextwidth]
  \column{0.47\textwidth}
  \textbf{Schools can change the choice set students perceive}
  \begin{itemize}
    \item counseling and information;
    \item academic preparation;
    \item expectations and confidence;
    \item peer norms and networks.
  \end{itemize}

  \column{0.47\textwidth}
  \textbf{A choice can be individually optimal and school-caused}
  \begin{itemize}
    \item beliefs about returns are endogenous;
    \item preparation affects feasible options;
    \item support affects search and applications.
  \end{itemize}
\end{columns}

\vspace{0.8em}
\begin{alertblock}{Interpretive implication}
Observed institution and field choices may partly reflect the information, expectations, and preparation produced by the high school attended.
\end{alertblock}
\end{frame}
\end{comment}

\begin{frame}{The empirical design proceeds in two steps}
\begin{wideitemize}
  \item \textbf{Estimate outcome-specific value added.} I link Chile's centralized high-school assignment records to enrollment histories, standardized achievement, and higher-education enrollment.

  \item \textbf{Validate it causally.} I estimate school value added in a broad national panel, then test its causal content using random tie-breakers in assignment.
\end{wideitemize}
\end{frame}

\begin{frame}{Higher-value-added schools change both achievement and college choices}
\begin{wideitemize}
  \item Lottery-induced attendance at higher-value-added schools improves math, verbal achievement, entry into high-premium institutions and fields, and projected program income.

  \item Moving from the 10th to the 90th percentile predicts a \key{0.435 SD} gain in math and a \key{0.057 log-point} gain in projected program income.

  \item Achievement value added does \textit{not} summarize the school dimensions that predict college-field choices and projected income.

  \item Math effects vary with baseline achievement; the higher-education choice effects do not.
\end{wideitemize}
\end{frame}


\section{Context and Data}

\begin{frame}{Chile links national school histories to later college choices}
\begin{wideitemize}
  \item Formal secondary education begins in grade 9 and lasts four years.

  \item Public and private-subsidized schools participate in centralized assignment; private non-subsidized schools account for about 9.3\% of the grade-8 universe.

  \item Scrambled student identifiers link annual enrollment, grades, attendance, standardized achievement, school preferences, lottery numbers, and offers.
\end{wideitemize}
\end{frame}

\begin{frame}{Higher-education applications reveal institution and field choices}
\begin{wideitemize}
  \item The national admissions route combines standardized tests, school GPA-based scores, and students' ranked program preferences.

  \item Admission occurs at the institution-program level, with each program choosing the weights placed on score components.

  \item Administrative records contain exam outcomes, applications, admission scores, and first observed enrollment.

  \item The linked data therefore capture whether students enroll, where they enroll, and which field they choose.
\end{wideitemize}
\end{frame}



\begin{frame}{Chile's centralized assignment system creates national lottery variation}
\begin{wideitemize}
  \item Families submit ranked school lists to the \textit{Sistema de Admisi\'on Escolar} (SAE), which uses student-proposing Deferred Acceptance.

  \item Priorities and vacancies determine admission; random tie-breakers resolve excess demand within priority groups.

  \item Conditional on priorities and simulated assignment risk, tie-breaker offers generate quasi-experimental variation. \parencite{abdulkadiroglu_research_2017}

  \item Replicating the mechanism with the original lottery numbers reproduces more than \key{99.5\%} of first-round assignments in every 2018--2021 cohort.
\end{wideitemize}
\end{frame}

\section{Methodology and Empirical Framework}

\begin{frame}{The challenge is sorting across many schools and fallback options}
\begin{columns}[T,onlytextwidth]
  \column{0.48\textwidth}
  \textbf{Selection}
  \begin{itemize}
    \item Families sort on resources, preferences, location, information, and expectations.
    \item These same factors predict later college choices.
  \end{itemize}

  \vspace{0.6em}
  \textbf{Many treatments}
  \[
    D_i^{\mathrm{full}}
    =
    \big(
    \mathbf{1}\{S_i=1\},\ldots,
    \mathbf{1}\{S_i=J\}
    \big).
  \]

  \column{0.48\textwidth}
  \textbf{Different counterfactual schools}
  \begin{itemize}
    \item An offer to school $A$ can move one student from $G^1$ and another from $G^2$.
    \item Fully flexible effects become target-school-by-fallback objects.
  \end{itemize}

  \[
    \beta_{AG^1},\ \beta_{AG^2},\ldots,
    \beta_{CG^K}.
  \]
\end{columns}

\vspace{0.5em}
\begin{center}
\small\key{I project high-school attendance onto outcome-specific scalar value-added indices.}
\end{center}
\end{frame}

\begin{frame}{The design estimates and then validates outcome-specific school value added}
\begin{columns}[c,onlytextwidth]
  \column{0.44\textwidth}
  \begin{block}{1. Broad national panel}
  Estimate observational school value added for each outcome using rich pre-treatment controls.
  \[
    \widehat V_j^m
  \]
  \end{block}

  \column{0.08\textwidth}
  \centering
  $\Longrightarrow$

  \column{0.44\textwidth}
  \begin{block}{2. SAE lottery sample}
  Test whether an offer to a higher-value-added school causally improves the same outcome.
  \[
    \theta^m
  \]
  \end{block}
\end{columns}

\vspace{1em}
\begin{center}
\large
Does the observational difference encoded in $\widehat V_j^m$ \textbf{pass through} when school attendance changes because of the lottery?
\end{center}
\end{frame}

\section{Empirical Framework}

\begin{frame}{Step 1: Estimate observational value added in the broad panel}
\[
  Y_i^m
  =
  \alpha^m
  \lambda_{s(i)}^m
  + X_i'\Gamma^m
  + u_i^m,
  \qquad
  \widehat V_j^m
  =
  \widehat\lambda_j^m-\overline{\widehat\lambda}^{\,m}_{\omega}.
\]

\small
\begin{itemize}
  \addtolength{\itemsep}{0.6em}
  \item Estimate a separate school fixed effect for math, verbal achievement, high-premium institution, high-premium field, and log projected income.

  \item $X_i$ includes grade-4 achievement polynomials, family background, middle-school grades and attendance, cohort, gender, age, and home location.

  \item These measures remain observational: sorting not absorbed by $X_i$ can contaminate $\widehat V_j^m$.

  \item The lottery step asks whether the measured school differences contain causal signal.
\end{itemize}
\end{frame}

\begin{frame}{Empirical Bayes shrinks less precise school estimates more strongly}
\small
\begin{columns}[c,onlytextwidth]
  \column{0.48\textwidth}
  \[
    \widehat\tau_m^2
    =
    \max\left\{
      \operatorname{Var}_{\omega}(\widehat V_j^m)
      -\sum_j\omega_j(se_j^m)^2,
      0
    \right\}
  \]

  \column{0.48\textwidth}
  \[
    \lambda_j^m
    =
    \frac{\widehat\tau_m^2}
    {\widehat\tau_m^2+(se_j^m)^2},
    \qquad
    \widehat V_j^{m,EB}
    =
    \lambda_j^m\widehat V_j^m.
  \]
\end{columns}

\vspace{0.9em}
\begin{wideitemize}
  \item Large, precisely estimated schools remain close to their original fixed effects.
  \item Small or noisy schools move toward the student-weighted mean of zero. \parencite{walters_chapter_2024}
  \item Shrinkage improves the signal-to-noise ratio of the index used in the lottery design; it does not create a separate causal estimator.
\end{wideitemize}
\end{frame}

\begin{frame}{Step 2: Collapse each student's assignment risk to expected value added}
\small
\begin{columns}[T,onlytextwidth]
  \column{0.44\textwidth}
  \textbf{Assignment portfolio}
  \[
    p_i=(p_{i1},\ldots,p_{iJ}),
    \qquad
    p_{ij}=\Pr(Z_i=j).
  \]
  Repeated simulations of the assignment mechanism recover each probability.

  \column{0.52\textwidth}
  \textbf{Outcome-specific risk control}
  \[
    E_i^{m,EB}
    =
    \sum_j p_{ij}\widehat V_j^{m,EB}.
  \]
  This is the school value added student $i$ expects over repeated lottery draws, holding the submitted application and priorities fixed.
\end{columns}

\vspace{1em}
\begin{alertblock}{Identification intuition}
The lottery offer's value added, relative to the portfolio's expected value added, is quasi-experimental conditional on assignment risk.
\end{alertblock}
\end{frame}

\begin{frame}{Offered-school value added instruments attended-school value added}
\begin{columns}[c,onlytextwidth]
  \column{0.29\textwidth}
  \centering
  \textbf{Lottery offer}
  \[
    O_i^{m,EB}
    =
    \widehat V_{Z_i}^{m,EB}
  \]
  \footnotesize Exogenous conditional on assignment risk

  \column{0.07\textwidth}
  \centering
  $\Longrightarrow$

  \column{0.29\textwidth}
  \centering
  \textbf{School attended}
  \[
    A_i^{m,EB}
    =
    \widehat V_{D_i}^{m,EB}
  \]
  \footnotesize Endogenous treatment

  \column{0.07\textwidth}
  \centering
  $\Longrightarrow$

  \column{0.28\textwidth}
  \centering
  \textbf{Student outcome}
  \[
    Y_i^m
  \]
  \footnotesize Same outcome used to define value added
\end{columns}

\vspace{1.3em}
\[
  Y_i^m
  =
  \alpha^m
  \theta^m A_i^{m,EB}
  + \rho^m E_i^{m,EB}
  + W_i'\delta^m
  + \eta_i^m,
  \qquad
  A_i^{m,EB}\ \text{instrumented by}\ O_i^{m,EB}.
\]
\end{frame}

\begin{frame}{The forecast coefficient measures causal pass-through}
\begin{center}
\Large
$\theta^m$
\end{center}

\vspace{0.4em}
\begin{columns}[T,onlytextwidth]
  \column{0.30\textwidth}
  \begin{block}{$\theta^m=0$}
  Observational value added has no causal predictive content along the lottery margin.
  \end{block}

  \column{0.30\textwidth}
  \begin{block}{$0<\theta^m<1$}
  Value added is directionally informative but overstates the lottery-induced causal gain.
  \end{block}

  \column{0.30\textwidth}
  \begin{block}{$\theta^m=1$}
  The observational difference passes through one-for-one into the causal outcome difference.
  \end{block}
\end{columns}

\vspace{0.8em}
\begin{center}
This is a validation design for school value added, following the forecast-coefficient logic of \textcite{angrist_leveraging_2017}.
\end{center}
\end{frame}

\section{Conclusion}

\begin{frame}{Achievement-based measures of school quality are incomplete}
\begin{itemize}
  \addtolength{\itemsep}{0.85em}
  \item Outcome-specific observational value added predicts lottery-induced causal gains in achievement, institution choice, field choice, and projected program income.

  \item Achievement pass-through varies by baseline achievement and gender, while higher-education pass-through is comparatively stable across these groups.

  \item College-choice value-added dimensions contain projected-income signal beyond math value added, so score-centered accountability can miss consequential school effects on fields and institutions.
\end{itemize}
\end{frame}

\begin{frame}[standout]
High schools shape not only how much students learn,\\[0.5em]
but also the options they perceive and choose.
\end{frame}

\begin{frame}{Different fallbacks make school-specific IV effects unwieldy}
\small
\begin{itemize}
  \addtolength{\itemsep}{0.45em}
  \item With a common guaranteed fallback $G$, lottery-induced comparisons could be written as
  \[
    \beta_{AG},\qquad \beta_{BG},\qquad \beta_{CG}.
  \]

  \item In practice, students' fallback schools differ:
  \[
    G_i\in\{G^1,G^2,\ldots,G^K\}.
  \]

  \item The relevant objects become target-school-by-fallback effects:
  \[
    \beta_{AG^1},\ \beta_{AG^2},\ldots,\beta_{CG^K}.
  \]

  \item The scalar value-added design instead asks whether lottery-induced movement toward schools with higher measured value causes better outcomes.
\end{itemize}
\end{frame}

\begin{frame}{Formal first stage}
\[
  A_i^{m,EB}
  =
  a^m
  \kappa^m O_i^{m,EB}
  + \rho_A^m E_i^{m,EB}
  + W_i'\delta_A^m
  + \nu_i^m.
\]

\begin{wideitemize}
  \item $A_i^{m,EB}$ is the EB value added of the attended high school.
  \item $O_i^{m,EB}$ is the EB value added of the first-round offered school.
  \item $E_i^{m,EB}$ is expected EB value added under the student's simulated assignment probabilities.
  \item $W_i$ contains cohort, gender, age, and grade-4 math and language controls.
\end{wideitemize}
\end{frame}

\begin{frame}{Ordered value-added IV}
\footnotesize
\vspace{0.4em}
\[
\begin{aligned}
Y_i^{PI}
= {}& \alpha
+ \theta_M A_i^M
+ \theta_I A_i^{I\perp M}
+ \theta_F A_i^{F\perp M,I} \\
&+ \rho_M E_i^M
+ \rho_I E_i^{I\perp M}
+ \rho_F E_i^{F\perp M,I}
+ W_i'\delta
+ \eta_i.
\end{aligned}
\]

\begin{itemize}
  \addtolength{\itemsep}{0.45em}
  \item $A_i^M$: math value added.
  \item $A_i^{I\perp M}$: institution value added residualized on math value added.
  \item $A_i^{F\perp M,I}$: field value added residualized on the first two dimensions.
  \item Each attended-school value is instrumented with the corresponding first-round offered-school value.
\end{itemize}
\end{frame}
